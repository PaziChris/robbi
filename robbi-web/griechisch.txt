#
# AUSSPRACHE:
#   á é í ó ú = diese Silbe BETONEN (sehr wichtig im Griechischen!)
#   ch = wie "ch" in "Bach" (vor e/i wie in "ich")
#   gh = weiches Rachen-g (wie ein gehauchtes r)
#   j  = wie "j" in "ja"
#   th = wie engl. "think"       dh = wie engl. "this"
#   z  = weiches s (wie "Sonne") s  = scharfes s
#   ts = wie dt. "z"             tz = wie "ds"
#   u  = u      i = i      e = e      o = o
#   Achtung: "ne" heißt JA, "óchi" heißt NEIN!
#   (m) = zu einem Mann / männlich, Frauenform steht in der Notiz
